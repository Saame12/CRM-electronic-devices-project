function surveyPage() {
  const counts=surveyWorkspace.counts || {ALL:surveys.length,DRAFT:surveys.filter(s=>s.status==='DRAFT').length,OPEN:surveys.filter(s=>s.status==='OPEN').length,CLOSED:surveys.filter(s=>s.status==='CLOSED').length};
  const total=surveyWorkspace.total ?? surveys.length, start=total?(surveyWorkspace.page-1)*surveyWorkspace.pageSize+1:0;
  return `<div class="survey-workspace"><div class="survey-list-heading"><div><h1>${surveyFilters.hidden?'Khảo sát đã ẩn':'Quản lý khảo sát'}</h1><p class="muted">Xem và thao tác trong khung riêng, giữ nguyên vị trí danh sách.</p></div><div class="toolbar"><button type="button" class="btn" data-action="toggle-hidden-surveys">${surveyFilters.hidden?'← Danh sách khảo sát':'Xem khảo sát đã ẩn'}</button>${!surveyFilters.hidden?'<button type="button" class="btn primary" data-action="show-create">Tạo khảo sát</button>':''}</div></div><nav class="survey-status-tabs" role="tablist" aria-label="Trạng thái khảo sát">${[['','Tất cả'],['OPEN','Đang mở'],['CLOSED','Đã đóng'],['DRAFT','Bản nháp']].map(([value,label])=>`<button type="button" class="btn" role="tab" aria-selected="${(surveyFilters.status||'')===value}" data-action="survey-tab" data-status="${value}" aria-controls="surveyList">${label} <span>${counts[value||'ALL']||0}</span></button>`).join('')}</nav><div class="survey-list-tools"><form id="surveySearchForm" role="search" class="toolbar"><label>Tìm khảo sát<input id="surveySearch" name="q" value="${esc(surveyFilters.q)}" placeholder="Tiêu đề hoặc mô tả"></label><button class="btn">Tìm kiếm</button></form><label>Số dòng mỗi trang<select id="surveyPageSize">${[10,20,50,100].map(n=>`<option value="${n}"${surveyWorkspace.pageSize===n?' selected':''}>${n}</option>`).join('')}</select></label></div><div id="surveyList" class="card table-wrap"><table class="table"><thead><tr><th>Khảo sát</th><th>Kết thúc</th><th>Trạng thái</th><th>Số người đã làm</th><th>Thao tác</th></tr></thead><tbody>${surveys.map(s=>{
    const actions=surveyRowActions(s),primary=actions[0];
    return `<tr data-survey-row="${esc(s.id)}"${surveyWorkspace.lastId===s.id?' class="survey-last-viewed"':''}><td><b>${esc(s.title)}</b><p class="survey-description">${esc(s.description)}</p><small class="muted">Phạm vi: ${esc(surveyAudienceSummary(s.audience_config))}</small></td><td>${s.due_date?esc(CrmDomain.iso(s.due_date).split('-').reverse().join('/')):'Đóng thủ công'}${s.auto_closed_at?'<br><small>Tự đóng do hết hạn</small>':''}</td><td><span class="badge ${s.status}">${surveyStatusLabel(s.status)}</span></td><td>${Number(s.response_count)||0}</td><td><div class="survey-row-actions">${surveyActionButton(s,primary)}<details class="survey-action-menu"><summary>Thao tác ⋯</summary><div>${actions.slice(1).map(action=>surveyActionButton(s,action)).join('')}</div></details></div></td></tr>`;
  }).join('') || '<tr><td colspan="5">Không có khảo sát phù hợp.</td></tr>'}</tbody></table></div><div class="survey-pagination"><p class="muted" role="status">${start}–${Math.min(start+surveys.length-1,total)} / ${total} khảo sát · Trang ${surveyWorkspace.page}/${surveyWorkspace.totalPages}</p><div><button type="button" class="btn" data-action="survey-page" data-page-number="${surveyWorkspace.page-1}"${surveyWorkspace.page<=1?' disabled':''}>← Trang trước</button> <button type="button" class="btn" data-action="survey-page" data-page-number="${surveyWorkspace.page+1}"${surveyWorkspace.page>=surveyWorkspace.totalPages?' disabled':''}>Trang sau →</button></div></div></div>`;
}
